\section{Search vs.\ decision vs.\ distinguishing.}
In post-quantum cryptography (PQC), many security assumptions are stated using one of three of hard problems:

\begin{enumerate}[label=\arabic*.]
	\item \textbf{Search problem:} recover the hidden object.
	\item \textbf{Decision problem:} answer a yes/no question about the instance.
	\item \textbf{Distinguishing problem:} tell whether samples come from one distribution or another.
\end{enumerate}
These three look similar, but they play different roles in theory and in cryptographic reductions.

\subsection{Vector spaces and Euclidean geometry}

%One must understand the ambient geometry in which lattices live. The relevant setting is Euclidean space \(\mathbb{R}^n\).

\defbox[Vectors in \(\mathbb{R}^n\)]{
\begin{definition}
A vector in \(\mathbb{R}^n\) is an ordered \(n\)-tuple
\[
\mathbf{x} = (x_1,x_2,\dots,x_n), \qquad x_i \in \mathbb{R}.
\]
The set of all such vectors forms a vector space over \(\mathbb{R}\).
\end{definition}}

\medskip

\defbox[Inner product]{\begin{definition}
The standard inner product of two vectors \(\mathbf{x},\mathbf{y} \in \mathbb{R}^n\) is defined by
\[
\langle \mathbf{x},\mathbf{y}\rangle
=
x_1y_1+x_2y_2+\cdots+x_ny_n.
\]
This quantity measures the geometric alignment of two vectors.	
\end{definition}}
\begin{remark}
Two vectors are said to be \emph{orthogonal} if
\[
\langle \mathbf{x},\mathbf{y}\rangle = 0.
\]
\end{remark}

\newpage
\defbox[Euclidean norm]{\begin{definition}
The Euclidean norm of a vector \(\mathbf{x}\in\mathbb{R}^n\) is
\[
\|\mathbf{x}\|
=
\sqrt{\langle \mathbf{x},\mathbf{x}\rangle}
=
\sqrt{x_1^2+x_2^2+\cdots+x_n^2}.
\]
\end{definition}}
\begin{remark}
This norm satisfies the standard properties:
\begin{enumerate}
	\item \(\|\mathbf{x}\|\ge 0\), with equality if and only if \(\mathbf{x}=\mathbf{0}\),
	\item \(\|\alpha \mathbf{x}\| = |\alpha|\,\|\mathbf{x}\|\) for \(\alpha \in \mathbb{R}\),
	\item \(\|\mathbf{x}+\mathbf{y}\| \le \|\mathbf{x}\|+\|\mathbf{y}\|\) (triangle inequality).
\end{enumerate}
\end{remark}

\medskip
\defbox[Linear independence and bases]{\begin{definition}
A finite set of vectors
\[
\{\mathbf{b}_1,\dots,\mathbf{b}_m\} \subseteq \mathbb{R}^n
\]
is \emph{linearly independent} if
\[
a_1\mathbf{b}_1+\cdots+a_m\mathbf{b}_m = \mathbf{0}
\]
implies
\[
a_1=\cdots=a_m=0.
\]
\end{definition}}
\begin{remark}
If \(\{\mathbf{b}_1,\dots,\mathbf{b}_m\}\) is linearly independent, then its real span
\[
V
=
\left\{
\sum_{i=1}^m a_i \mathbf{b}_i : a_i \in \mathbb{R}
\right\}
\]
is an \(m\)-dimensional subspace of \(\mathbb{R}^n\), and the vectors \(\mathbf{b}_1,\dots,\mathbf{b}_m\) form a basis of \(V\).
\end{remark}
\begin{remark}
If \(B\) is the matrix whose columns are \(\mathbf{b}_1,\dots,\mathbf{b}_m\), then
\[
V = \{B\mathbf{a} : \mathbf{a}\in\mathbb{R}^m\}.
\]	
\end{remark}
