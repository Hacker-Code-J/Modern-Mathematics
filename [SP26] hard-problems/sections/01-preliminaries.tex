\section{Search vs.\ decision vs.\ distinguishing.}
In post-quantum cryptography (PQC), many security assumptions are stated using one of three of hard problems:

\begin{enumerate}[label=\arabic*.]
	\item \textbf{Search problem:} recover the hidden object.
	\item \textbf{Decision problem:} answer a yes/no question about the instance.
	\item \textbf{Distinguishing problem:} tell whether samples come from one distribution or another.
\end{enumerate}
These three look similar, but they play different roles in theory and in cryptographic reductions.

\subsection{Vector spaces and Euclidean geometry}

%One must understand the ambient geometry in which lattices live. The relevant setting is Euclidean space \(\mathbb{R}^n\).

\defbox[Vectors in \(\mathbb{R}^n\)]{
\begin{definition}
A vector in \(\mathbb{R}^n\) is an ordered \(n\)-tuple
\[
\mathbf{x} = (x_1,x_2,\dots,x_n), \qquad x_i \in \mathbb{R}.
\]
The set of all such vectors forms a vector space over \(\mathbb{R}\).
\end{definition}}

\medskip

\defbox[Inner product]{\begin{definition}
The standard inner product of two vectors \(\mathbf{x},\mathbf{y} \in \mathbb{R}^n\) is defined by
\[
\langle \mathbf{x},\mathbf{y}\rangle
=
x_1y_1+x_2y_2+\cdots+x_ny_n.
\]
This quantity measures the geometric alignment of two vectors.	
\end{definition}}
\begin{remark}
Two vectors are said to be \emph{orthogonal} if
\[
\langle \mathbf{x},\mathbf{y}\rangle = 0.
\]
\end{remark}

\newpage
\defbox[Euclidean norm]{\begin{definition}
The Euclidean norm of a vector \(\mathbf{x}\in\mathbb{R}^n\) is
\[
\|\mathbf{x}\|
=
\sqrt{\langle \mathbf{x},\mathbf{x}\rangle}
=
\sqrt{x_1^2+x_2^2+\cdots+x_n^2}.
\]
\end{definition}}
\begin{remark}
This norm satisfies the standard properties:
\begin{enumerate}
	\item \(\|\mathbf{x}\|\ge 0\), with equality if and only if \(\mathbf{x}=\mathbf{0}\),
	\item \(\|\alpha \mathbf{x}\| = |\alpha|\,\|\mathbf{x}\|\) for \(\alpha \in \mathbb{R}\),
	\item \(\|\mathbf{x}+\mathbf{y}\| \le \|\mathbf{x}\|+\|\mathbf{y}\|\) (triangle inequality).
\end{enumerate}
\end{remark}

\medskip
\defbox[Linear Independence]{\begin{definition}
A finite set of vectors
\[
\{\mathbf{b}_1,\dots,\mathbf{b}_m\} \subseteq \mathbb{R}^n
\]
is \textbf{linearly independent} if
\[
a_1\mathbf{b}_1+\cdots+a_m\mathbf{b}_m = \mathbf{0}
\]
implies
\[
a_1=\cdots=a_m=0.
\]
\end{definition}}
\begin{remark}[span]
The real span of \(\mathbf b_1,\dots,\mathbf b_m\) is
\[
V=\textrm{span}_\R(\textbf{b}_1,\dots,\textbf{b}_m)=\left\{\sum_{i=1}^m a_i\mathbf b_i : a_i\in\mathbb R\right\}.
\]
So \(V\) is just the set of all vectors you can build from \(\mathbf b_1,\dots,\mathbf b_m\) using real coefficients.
\end{remark}

\medskip

\begin{remark}
Let \(B\) be the matrix whose columns are \(\mathbf b_1,\dots,\mathbf b_m\):
\[
B=
\begin{bmatrix}
	| & & |\\
	\mathbf b_1 & \cdots & \mathbf b_m\\
	| & & |
\end{bmatrix}.
\] Then
\[
B\mathbf a
=\begin{bmatrix}
	| & & |\\
	\mathbf b_1 & \cdots & \mathbf b_m\\
	| & & |
\end{bmatrix}\begin{bmatrix}
	a_1\\
	\vdots\\
	a_m
\end{bmatrix}=
a_1\begin{bmatrix}
	|\\  \mathbf b_1 \\|
\end{bmatrix}+
\cdots+
a_m\begin{bmatrix}
	|\\  \mathbf b_m \\|
\end{bmatrix}
=
a_1\mathbf b_1+\cdots+a_m\mathbf b_m.,
\]
So
\[
V=\operatorname{span}_{\mathbb R}\{\mathbf b_1,\dots,\mathbf b_m\}
=
\left\{\sum_{i=1}^m a_i\mathbf b_i:a_i\in\mathbb R\right\}
=
\{B\mathbf a:\mathbf a\in\mathbb R^m\}.
\]
The first writes the span as a sum.  
The second writes the same thing using matrix multiplication.
\end{remark}

\newpage
\defbox[Lattice]{
\begin{definition}
	A lattice \(\Lambda \subset \mathbb R^n\) is called a lattice of rank \(m\) if
	\[
	\Lambda=
	\left\{
	\sum_{i=1}^m z_i \mathbf b_i : z_i\in\mathbb Z
	\right\},
	\]
	where \(\mathbf b_1,\dots,\mathbf b_m\in\mathbb R^n\) are linearly independent.
\end{definition}}
\begin{remark}
\; \begin{itemize}
	\item If \(m=n\), then \(\Lambda\) is called a full-rank lattice.  
	\item If \(m<n\), then \(\Lambda\) is a non-full-rank lattice, and it lies in the
	\(m\)-dimensional subspace
	\[
	\operatorname{span}_{\mathbb R}\{\mathbf b_1,\dots,\mathbf b_m\}\subsetneq \mathbb R^n.
	\]
\end{itemize}
\end{remark}

\medskip

\begin{remark}[Full-rank Lattice]
A \textbf{full-rank lattice} \(\Lambda \subset \mathbb{R}^n\) is a set of the form
\[
\Lambda=\mathcal{L}(B)=\{B\mathbf z:\mathbf z\in\mathbb Z^n\},
\]
where \(B\in\mathbb R^{n\times n}\) is an invertible matrix. If $
B=
\begin{bmatrix}
	\mathbf b_1 & \mathbf b_2 & \cdots & \mathbf b_n
\end{bmatrix}$, 
then equivalently
\[
\mathcal{L}(B)=
\left\{
z_1\mathbf b_1+\cdots+z_n\mathbf b_n
:
z_1,\dots,z_n\in\mathbb Z
\right\}.
\]
Thus, a lattice is the set of all integer linear combinations of \(n\) linearly independent vectors in \(\mathbb R^n\).	
\end{remark}

%	The condition that \(B\) is invertible means that its columns \(\mathbf b_1,\dots,\mathbf b_n\) are linearly independent, so they form a basis of \(\mathbb R^n\). The word ``full-rank'' means that the lattice spans all of \(\mathbb R^n\), not just a lower-dimensional subspace.
%\end{note}

\subsubsection{Fundamental Parallelepiped}

Let
\[
B=
\begin{bmatrix}
	| & | & \cdots & | \\
	\mathbf b_1 & \mathbf b_2 & \cdots & \mathbf b_n \\
	| & | & \cdots & | \\
\end{bmatrix}
\in \mathbb R^{n\times n}
\]
be a basis matrix for a full-rank lattice
\[
\Lambda=\mathcal L(B)=\{B\mathbf z:\mathbf z\in\mathbb Z^n\}.
\]
Thus the columns \(\mathbf b_1,\dots,\mathbf b_n\in\mathbb R^n\) are linearly independent, and every lattice point is an integer linear combination of these basis vectors.

\medskip

\defbox[Fundamental Parallelepiped]{
\begin{definition}
The \textbf{fundamental parallelepiped} associated with \(B\) is defined by \[
\mathcal P(B)
=
\left\{
\sum_{i=1}^n \alpha_i \mathbf b_i : 0\le \alpha_i<1
\right\}.
\] Equivalently, in matrix form,
\[
\mathcal P(B)=\{B\boldsymbol\alpha:\boldsymbol\alpha\in[0,1)^n\}.
\]
\end{definition}}

\probox[Tiling Property]{\begin{proposition}
Let \(B=[\mathbf b_1\ \cdots\ \mathbf b_n]\in\mathbb R^{n\times n}\) be invertible, and let
\[
\Lambda=\mathcal L(B)=\{B\mathbf z:\mathbf z\in\mathbb Z^n\}
\]
be the lattice generated by the columns of \(B\). Define the fundamental parallelepiped
\[
\mathcal P(B)
=
\left\{
\sum_{i=1}^n \alpha_i \mathbf b_i : 0\le \alpha_i<1
\right\}
=
\{B\boldsymbol\alpha:\boldsymbol\alpha\in[0,1)^n\}.
\]
Then:
\begin{enumerate}
	\item \(\mathbb R^n\) is covered by translates of \(\mathcal P(B)\) by lattice vectors:
	\[
	\mathbb R^n=\bigcup_{\mathbf y\in\Lambda}(\mathbf y+\mathcal P(B)).
	\]
	\item If \(\mathbf y_1,\mathbf y_2\in\Lambda\) with \(\mathbf y_1\neq \mathbf y_2\), then
	\[
	(\mathbf y_1+\mathcal P(B))\cap(\mathbf y_2+\mathcal P(B))=\varnothing.
	\]
\end{enumerate}

\begin{proof}
	Since \(B\) is invertible, for every \(\mathbf x\in\mathbb R^n\) there exists a unique vector \(\mathbf t\in\mathbb R^n\) such that
	\[
	\mathbf x=B\mathbf t.
	\]
	Write
	\[
	\mathbf t=
	\begin{pmatrix}
		t_1\\
		\vdots\\
		t_n
	\end{pmatrix}.
	\]
	For each \(i\), let
	\[
	z_i=\lfloor t_i\rfloor\in\mathbb Z,
	\qquad
	\alpha_i=t_i-z_i.
	\]
	Then
	\[
	0\le \alpha_i<1
	\qquad\text{for each }i,
	\]
	and
	$t_i=z_i+\alpha_i$. Hence
	\[
	\mathbf t=\mathbf z+\boldsymbol\alpha,\quad\text{where}\quad
	\mathbf z=
	\begin{pmatrix}
		z_1\\
		\vdots\\
		z_n
	\end{pmatrix}\in\mathbb Z^n,
	\qquad
	\boldsymbol\alpha=
	\begin{pmatrix}
		\alpha_1\\
		\vdots\\
		\alpha_n
	\end{pmatrix}\in[0,1)^n.
	\]
	Multiplying by \(B\), we obtain
	\[
	\mathbf x=B\mathbf t=B\mathbf z+B\boldsymbol\alpha.
	\]
	Now \(B\mathbf z\in\Lambda\), and \(B\boldsymbol\alpha\in\mathcal P(B)\). Therefore
	\[
	\mathbf x\in B\mathbf z+\mathcal P(B)\subseteq \bigcup_{\mathbf y\in\Lambda}(\mathbf y+\mathcal P(B)).
	\]
	Since \(\mathbf x\in\mathbb R^n\) was arbitrary, it follows that
	\[
	\mathbb R^n=\bigcup_{\mathbf y\in\Lambda}(\mathbf y+\mathcal P(B)).
	\]
	It remains to prove disjointness. Suppose
	\[
	\mathbf x\in(\mathbf y_1+\mathcal P(B))\cap(\mathbf y_2+\mathcal P(B))
	\]
	for some \(\mathbf y_1,\mathbf y_2\in\Lambda\). Then there exist
	\[
	\boldsymbol\alpha,\boldsymbol\beta\in[0,1)^n
	\]
	such that
	\[
	\mathbf x=\mathbf y_1+B\boldsymbol\alpha=\mathbf y_2+B\boldsymbol\beta.
	\]
	Since \(\mathbf y_1,\mathbf y_2\in\Lambda\), there exist \(\mathbf z_1,\mathbf z_2\in\mathbb Z^n\) such that
	\[
	\mathbf y_1=B\mathbf z_1,
	\qquad
	\mathbf y_2=B\mathbf z_2.
	\]
	Therefore
	\[
	B\mathbf z_1+B\boldsymbol\alpha
	=
	B\mathbf z_2+B\boldsymbol\beta,
	\]
	so
	\[
	B(\mathbf z_1-\mathbf z_2)=B(\boldsymbol\beta-\boldsymbol\alpha).
	\]
	Because \(B\) is invertible,
	\[
	\mathbf z_1-\mathbf z_2=\boldsymbol\beta-\boldsymbol\alpha.
	\]
%	The left-hand side is an integer vector, while each coordinate of the right-hand side lies in the interval \((-1,1)\), since \(\alpha_i,\beta_i\in[0,1)\). 
	The only integer in \((-1,1)\) is \(0\), so
	\[
	\boldsymbol\beta-\boldsymbol\alpha=\mathbf 0.
	\]
	Hence
	\[
	\boldsymbol\alpha=\boldsymbol\beta
	\qquad\text{and}\qquad
	\mathbf z_1=\mathbf z_2.
	\]
	Therefore
	\[
	\mathbf y_1=B\mathbf z_1=B\mathbf z_2=\mathbf y_2.
	\]
	This proves that if \(\mathbf y_1\neq \mathbf y_2\), then
	\[
	(\mathbf y_1+\mathcal P(B))\cap(\mathbf y_2+\mathcal P(B))=\varnothing.
	\]
%	Thus the translates of \(\mathcal P(B)\) by lattice vectors form a partition of \(\mathbb R^n\), and \(\mathcal P(B)\) is a fundamental domain for \(\Lambda\).	
%	Finally, if one uses the closed parallelepiped \(\overline{\mathcal P}(B)\) instead of \(\mathcal P(B)\), the same argument shows that every point of \(\mathbb R^n\) still lies in some translate, but uniqueness may fail when one or more coordinates are equal to \(0\) or \(1\). These are precisely boundary points of the parallelepiped, so distinct translates can intersect only along their boundaries.
\end{proof}

\thmbox[Uniqueness of Representation Modulo the Lattice]{\begin{theorem}
Let \(B\in\mathbb R^{n\times n}\) be invertible, and let
\[
\Lambda=\mathcal L(B)=\{B\mathbf z:\mathbf z\in\mathbb Z^n\}
\]
be the lattice generated by the columns of \(B\). Let
\[
\mathcal P(B)=\{B\boldsymbol\alpha:\boldsymbol\alpha\in[0,1)^n\}
\]
be the associated fundamental parallelepiped. Then every vector \(\mathbf x\in\mathbb R^n\) can be written uniquely in the form
\[
\mathbf x=\mathbf y+\mathbf p,
\qquad
\mathbf y\in\mathcal L(B),\ \mathbf p\in\mathcal P(B).
\]
Equivalently, for every \(\mathbf x\in\mathbb R^n\), there exist unique $
\mathbf z\in\mathbb Z^n$, $\boldsymbol\alpha\in[0,1)^n$ such that
\[
\mathbf x=B\mathbf z+B\boldsymbol\alpha=B(\textbf{z}+\boldsymbol{\alpha}).
\]
%	Every vector \(\mathbf x\in\mathbb R^n\) can be written uniquely in the form
%	\[
%	\mathbf x=\mathbf y+\mathbf p,
%	\qquad
%	\mathbf y\in\Lambda,\ \mathbf p\in\mathcal P(B).
%	\]
%	Indeed, since \(B\) is invertible, we may write
%	\[
%	\mathbf x=B\mathbf t
%	\qquad\text{for some }\mathbf t\in\mathbb R^n.
%	\]
%	Writing each coordinate as
%	\[
%	t_i=z_i+\alpha_i,
%	\qquad
%	z_i\in\mathbb Z,\ 0\le \alpha_i<1,
%	\]
%	gives
%	\[
%	\mathbf x
%	=
%	B\mathbf z+B\boldsymbol\alpha=B(\textbf{z}+\boldsymbol{\alpha}),
%	\]
%	where \(B\mathbf z\in\Lambda\) and \(B\boldsymbol\alpha\in\mathcal P(B)\).
\end{theorem}}
\begin{proof}
	Let \(\mathbf x\in\mathbb R^n\) be arbitrary. Since \(B\) is invertible, there exists a unique vector
	\[
	\mathbf t=
	\begin{pmatrix}
		t_1\\
		\vdots\\
		t_n
	\end{pmatrix}
	\in\mathbb R^n
	\]
	such that $
	\mathbf x=B\mathbf t$. 
	
	\medskip
	\noindent
	(Existence) For each coordinate \(t_i\), define
	\[
	z_i=\lfloor t_i\rfloor\in\mathbb Z,
	\qquad
	\alpha_i=t_i-z_i.
	\]
	Then
	\[
	0\le \alpha_i<1
	\qquad\text{for each }i,
	\]
	and $
	t_i=z_i+\alpha_i.$ Hence
	\[
	\mathbf t=\mathbf z+\boldsymbol\alpha,\quad\text{where}\quad
	\mathbf z=
	\begin{pmatrix}
		z_1\\
		\vdots\\
		z_n
	\end{pmatrix}\in\mathbb Z^n,
	\qquad
	\boldsymbol\alpha=
	\begin{pmatrix}
		\alpha_1\\
		\vdots\\
		\alpha_n
	\end{pmatrix}\in[0,1)^n.
	\]
	Multiplying by \(B\), we obtain
	\[
	\mathbf x=B\mathbf t=B(\mathbf z+\boldsymbol\alpha)=B\mathbf z+B\boldsymbol\alpha.
	\]
	Now \(B\mathbf z\in\Lambda\) by definition of the lattice, and \(B\boldsymbol\alpha\in\mathcal P(B)\) by definition of the fundamental parallelepiped. Therefore
	\[
	\mathbf x=\mathbf y+\mathbf p
	\]
	with
	\[
	\mathbf y=B\mathbf z\in\Lambda,
	\qquad
	\mathbf p=B\boldsymbol\alpha\in\mathcal P(B).
	\]
	\newpage\noindent
	(Uniqueness)\; Suppose that
	\[
	\mathbf x=\mathbf y_1+\mathbf p_1=\mathbf y_2+\mathbf p_2,
	\]
	where
	\[
	\mathbf y_1,\mathbf y_2\in\Lambda,
	\qquad
	\mathbf p_1,\mathbf p_2\in\mathcal P(B).
	\]
	Since \(\mathbf y_1,\mathbf y_2\in\Lambda\), there exist \(\mathbf z_1,\mathbf z_2\in\mathbb Z^n\) such that
	\[
	\mathbf y_1=B\mathbf z_1,
	\qquad
	\mathbf y_2=B\mathbf z_2.
	\]
	Since \(\mathbf p_1,\mathbf p_2\in\mathcal P(B)\), there exist \(\boldsymbol\alpha,\boldsymbol\beta\in[0,1)^n\) such that
	\[
	\mathbf p_1=B\boldsymbol\alpha,
	\qquad
	\mathbf p_2=B\boldsymbol\beta.
	\]
	Thus
	\[
	B\mathbf z_1+B\boldsymbol\alpha
	=
	B\mathbf z_2+B\boldsymbol\beta.
	\]
	Rearranging gives
	\[
	B(\mathbf z_1-\mathbf z_2)=B(\boldsymbol\beta-\boldsymbol\alpha).
	\]
	Since \(B\) is invertible,
	\[
	\mathbf z_1-\mathbf z_2=\boldsymbol\beta-\boldsymbol\alpha.
	\]
%	The left-hand side is an integer vector, while each coordinate of the right-hand side lies in the interval \((-1,1)\), because \(\alpha_i,\beta_i\in[0,1)\). 
	The only integer in \((-1,1)\) is \(0\), so
	\[
	\boldsymbol\beta-\boldsymbol\alpha=\mathbf 0.
	\]
	Hence
	\[
	\boldsymbol\alpha=\boldsymbol\beta
	\qquad\text{and}\qquad
	\mathbf z_1=\mathbf z_2.
	\]
	Therefore
	\[
	\mathbf y_1=B\mathbf z_1=B\mathbf z_2=\mathbf y_2
	\]
	and
	\[
	\mathbf p_1=B\boldsymbol\alpha=B\boldsymbol\beta=\mathbf p_2.
	\]
	So the representation is unique. 
%	Thus every \(\mathbf x\in\mathbb R^n\) admits a unique decomposition
%	\[
%	\mathbf x=\mathbf y+\mathbf p,
%	\qquad
%	\mathbf y\in\Lambda,\ \mathbf p\in\mathcal P(B).
%	\]
\end{proof}
